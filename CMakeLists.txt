    set(HWY_ENABLE_TESTS OFF)

function(add_jxl_target target type)
    add_library(${target} ${type} ${ARGN})
    target_compile_features(${target} PUBLIC cxx_std_17)
    target_include_directories(${target} PUBLIC
        "$<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/lib/include>"
        "$<BUILD_INTERFACE:${CMAKE_CURRENT_BINARY_DIR}/include>"
        "$<INSTALL_INTERFACE:${CMAKE_INSTALL_INCLUDEDIR}>"
        PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}")
    set_target_properties(${target} PROPERTIES POSITION_INDEPENDENT_CODE ON
        CXX_VISIBILITY_PRESET hidden VISIBILITY_INLINES_HIDDEN ON)
    if(JXL_INSTALL)
        install(TARGETS ${target})
    endif()
endfunction()

function(add_jxl_variant type suffix)
    set(codec "jxl${suffix}")
    set(threads "jxl_threads${suffix}")
    set(cms "jxl_cms${suffix}")
    add_jxl_target(${codec} ${type} ${codec_sources})
    add_jxl_target(${threads} ${type} ${thread_sources})
    add_jxl_target(${cms} ${type} ${cms_sources} ${skcms_sources})
    target_include_directories(${cms} PRIVATE third_party/skcms)
    target_compile_definitions(${cms} PRIVATE ${skcms_definitions})
    target_link_libraries(${codec} PUBLIC ${cms} PRIVATE hwy brotlienc brotlidec brotlicommon)
    target_link_libraries(${cms} PRIVATE hwy)
    target_link_libraries(${threads} PRIVATE Threads::Threads)
    if(type STREQUAL "STATIC")
        target_compile_definitions(${codec} PUBLIC JXL_STATIC_DEFINE)
        target_compile_definitions(${threads} PUBLIC JXL_THREADS_STATIC_DEFINE)
        target_compile_definitions(${cms} PUBLIC JXL_CMS_STATIC_DEFINE)
        if(WIN32)
            set_target_properties(${codec} PROPERTIES OUTPUT_NAME jxl_static)
            set_target_properties(${threads} PROPERTIES OUTPUT_NAME jxl_threads_static)
            set_target_properties(${cms} PROPERTIES OUTPUT_NAME jxl_cms_static)
        endif()
    else()
        set_target_properties(${codec} PROPERTIES OUTPUT_NAME jxl VERSION ${PROJECT_VERSION})
        set_target_properties(${threads} PROPERTIES OUTPUT_NAME jxl_threads VERSION ${PROJECT_VERSION})
        set_target_properties(${cms} PROPERTIES OUTPUT_NAME jxl_cms VERSION ${PROJECT_VERSION})
    endif()
endfunction()
